function fnExcelReport(tableid)
{


	var tab_text=  "<table border='2px'><tr bgcolor='#87AFC6'>";
    var textRange; var j=0;
    tab = document.getElementById(tableid);
    for(j = 0 ; j < tab.rows.length ; j++)
    {
    tab_text=tab_text+tab.rows[j].innerHTML+"</tr>";
    }
    tab_text=tab_text+"</table>";
    tab_text= tab_text.replace(/<A[^>]*>|<\/A>/g, "");
    tab_text= tab_text.replace(/<img[^>]*>/gi,"");
    tab_text= tab_text.replace(/<input[^>]*>|<\/input>/gi, "");
    var ua = window.navigator.userAgent;
    var msie = ua.indexOf("MSIE ");
    if (msie > 0 || !!navigator.userAgent.match(/Trident.*rv\:11\./))
    {
    txtArea1.document.open("txt/html","replace");
    txtArea1.document.write(tab_text);
    txtArea1.document.close();
    txtArea1.focus();
    sa=txtArea1.document.execCommand("SaveAs",true,"SearchPermitResult.xls");
    }
    else
    sa = window.open('data:application/vnd.ms-excel,' + encodeURIComponent(tab_text));
    return (sa);
}

function f_setsubtype() {


  if (xmlHttp.readyState == 4 && xmlHttp.status == 200)
  {
    var xmlDoc = xmlHttp.responseXML;
    var resultObj = xmlDoc.getElementsByTagName("result")[0];
    var strOption1 = "";
	var subOption ="";
	var delimiterPosition = 0;

	if (resultObj)
	{
	  var result = resultObj.firstChild.nodeValue;
      if (result == "true"){
		subOption = xmlDoc.getElementsByTagName("subOption")[0].firstChild.nodeValue;
		delimiterPosition = subOption.indexOf("###");
		while (delimiterPosition > 0) {
          strOption1 +=  '<OPTION value="' + unescape(subOption.substring(0,delimiterPosition)) + '">';
	      subOption = subOption.substring(delimiterPosition+3);
		  delimiterPosition = subOption.indexOf("###");
          strOption1 +=  subOption.substring(0,delimiterPosition) + '</OPTION> ';
	      subOption = subOption.substring(delimiterPosition+3);
		  delimiterPosition = subOption.indexOf("###");
        }
 	  }
      if (navigator.appName == "Microsoft Internet Explorer") {
        document.getElementById("SubType").outerHTML = '<select name="SubType" id="SubType"  OnChange="f_getworktype(this.value);" style="width:190px;"> <option value="" selected></option> ' + strOption1 + ' </select>';
	    document.getElementById("WorkType").outerHTML = '<select name="WorkType" id="WorkType" style="width:190px;"> <option value="" selected></option> </select>';
	  } else {
	  document.getElementById("SubType").innerHTML = '<option value="" selected></option> ' + strOption1;
	  document.getElementById("WorkType").innerHTML = '<option value="" selected></option>';
	  }
    }
    else
	{
     document.getElementById("SubType").outerHTML = '<select name="SubType" id="SubType" style="width:190px;"> <option value="" selected></option> </select>';

	    return;
    }
  }
}
